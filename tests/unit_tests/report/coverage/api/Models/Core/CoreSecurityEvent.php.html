185    /**
186     * @description Encodes the event context as JSON of at most CONTEXT_MAX_LENGTH characters.
187     *   An over-long context is shortened VALUE by value (every string cut to
188     *   CONTEXT_VALUE_MAX_LENGTH) instead of cutting the JSON text - see bugfix-note below.
189     * @param array $context Context from record().
190     * @return string|null JSON, or null for an empty / not encodable context.
191     * @bugfix-note (2026-10-22) The JSON used to be cut with mb_substr() at 500 characters,
192     *   which leaves invalid JSON - the model's `array` cast then returned null and the admin
193     *   saw no context at all for exactly the long (most suspicious) requests. Found by the
194     *   unit tests.
195     */
196    private static function encodeContext(array $context): ?string
197    {
198        if (empty($context)) {
199            return null;
200        }
201
202        $encoded = json_encode($context, JSON_UNESCAPED_UNICODE);
203        if ($encoded === false) {
204            return null;
205        }
206        if (mb_strlen($encoded) <= self::CONTEXT_MAX_LENGTH) {
207            return $encoded;
208        }
209
210        $shortened = array_map(
211            fn ($value) => is_string($value) ? self::limit($value, self::CONTEXT_VALUE_MAX_LENGTH) : $value,
212            $context
213        );
214        $encoded = json_encode($shortened, JSON_UNESCAPED_UNICODE);
215
216        return mb_strlen($encoded) <= self::CONTEXT_MAX_LENGTH ? $encoded : json_encode(['_truncated' => true]);
217    }
