421    public function refresh(Request $request): JsonResponse
422    {
423        $refreshToken = $request->input('refreshToken');
424        if (!$refreshToken) {
425            return response()->json(['message' => 'Refresh token is missing.'], 401);
426        }
427
428        $hashedRefreshToken = hash('sha256', $refreshToken);
429        $dbRefreshToken = RefreshToken::with('user')
430                                     ->where('token', $hashedRefreshToken)
431                                     ->where('expires_at', '>', now())
432                                     ->first();
433
434        if (!$dbRefreshToken || !$dbRefreshToken->user) {
435            // 'info', ne 'warning' - vypršelý refresh token je běžný legitimní jev
436            // (uživatel se dlouho nepřihlásil), ne sám o sobě signál útoku. Opakovaný
437            // vysoký objem ze stejné IP je i tak vidět přes `occurrences` v adminu.
438            CoreSecurityEvent::record(
439                'refresh_token_invalid',
440                'info',
441                $request->ip(),
442                CoreSecurityEvent::contextFromRequest($request)
443            );

467                'warning',
468                $request->ip(),

470            );
471            return response()->json(['message' => self::LOGIN_NOT_ALLOWED_MESSAGE], 403);
472        }
473
474        $dbRefreshToken->delete();
475        $user->tokens()->delete();
476
477        $newAccessToken = $user->createToken('access-token', ['*'], now()->addMinutes(60))->plainTextToken;
478        $newRefreshToken = Str::random(60);
479
480        RefreshToken::create([
481            'user_id'    => $user->id,
482            'token'      => hash('sha256', $newRefreshToken),
483            'expires_at' => now()->addDays(7),
484        ]);
485
486        return response()->json([
487            'token'        => $newAccessToken,
488            'refreshToken' => $newRefreshToken,
489        ], 200);
490    }
