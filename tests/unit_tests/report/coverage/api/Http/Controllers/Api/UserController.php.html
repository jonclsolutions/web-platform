299     * @note The e-mail access policy (`assertEmailDomainAllowed()`) is checked only

302     * edited as long as the e-mail stays the same.
303     */
304    public function update(UpdateUserRequest $request, string $id): JsonResponse

313        $targetIsSysadmin = $user->roles()->where('role_name', self::SYSADMIN_ROLE_NAME)->exists();
314        $promotingToSysadmin = isset($validated['role_id']) && !$targetIsSysadmin
315            && $this->isSysadminRoleId((int) $validated['role_id']);

322            return response()->json(['message' => 'Only another sysadmin may edit, or promote an account to, the sysadmin role.'], 403);
323        }

329        $isSelfEdit = (int) $request->user()->id === (int) $user->id;

331        // Login e-mail: whoever controls it controls the account (password reset and
332        // 2FA codes are delivered there), so only a sysadmin may change someone else's.
333        if (array_key_exists('user_email', $validated)
334            && $validated['user_email'] !== $user->user_email
335            && !$isSelfEdit
336            && !$actorIsSysadmin) {
337            $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to change another account's login e-mail (not sysadmin): {$user->user_email}", (int) $id, 'User');
338            return response()->json(['message' => "Only sysadmin may change another account's login e-mail."], 403);
339        }
340
341        // Block / unblock: both directions. Unblocking matters as much as blocking -
342        // otherwise a compromised account could simply unblock one that was just locked.
343        if (array_key_exists('is_blocked', $validated)
344            && (bool) $validated['is_blocked'] !== (bool) $user->is_blocked
345            && !$actorIsSysadmin) {
346            $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to block or unblock an account (not sysadmin): {$user->user_email}", (int) $id, 'User');
347            return response()->json(['message' => 'Only sysadmin may block or unblock an account.'], 403);
348        }
349
350        // E-mail access policy: a NEW login e-mail must be allowed, whoever sets it
351        // (sysadmin and self-edit included). Compared case-insensitively, so sending the
352        // stored value back unchanged never triggers it - see the @note above.

354            && strcasecmp(trim((string) $validated['user_email']), (string) $user->user_email) !== 0) {

363        // requested change and would make this comparison always false).
364        //

373        // different `role_id` (which was always going to be silently ignored)

375        // even though the account's role never actually changed.
376        $oldRoleId = $user->roles()->first()?->id;
377        $roleWillActuallyApply = $request->user()->id !== $user->id;
378        $roleChanged = $roleWillActuallyApply

380            && (int) $validated['role_id'] !== (int) $oldRoleId;

382        // Own role: refused explicitly instead of being silently ignored, so the caller is
383        // never told "saved" for a change that did not happen. Sending the current role back
384        // unchanged (the admin form sends the whole record) is still fine.
385        if (!$roleWillActuallyApply
386            && isset($validated['role_id'])

390        }
391
392        if ($roleChanged && !$this->actorCanAssignRole($request, (int) $validated['role_id'])) {
393            $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to assign a role granting more than the actor's own permissions: {$user->user_email}", (int) $id, 'User');
394            return response()->json(['message' => 'You cannot assign a role that grants permissions you do not yourself have.'], 403);
395        }

397        // ── 2FA enforcement (role) - see refactor-note in the header ─────────────────
398        // @bugfix-note (2026-09-06) Uses $oldRoleId whenever the role change won't
399        // actually apply (self-edit) - same principle as $roleChanged above. Without
400        // this, a self-edit that merely INCLUDED a different (never-applied) role_id
401        // would evaluate 2FA enforcement, the is_blocked protected-role check, and
402        // applyExplicitPermissions()'s redundant-permission filtering against a role
403        // the account doesn't actually end up with.
404        $effectiveRoleId = $roleWillActuallyApply ? ($validated['role_id'] ?? $oldRoleId) : $oldRoleId;
405        $forcedByRole = $this->isRoleForced2fa($effectiveRoleId);
406
407        // Effective enforcement = role OR an existing sysadmin override (if the
408        // request does not change two_fa_forced_by_admin, the existing DB value is
409        // used).
410        $effectiveAdminForced = array_key_exists('two_fa_forced_by_admin', $validated)
411            ? (bool) $validated['two_fa_forced_by_admin']
412            : (bool) $user->two_fa_forced_by_admin;
413
414        $isForced = $forcedByRole || $effectiveAdminForced;
415
416        if ($isForced && array_key_exists('enable_2fa', $validated) && !$validated['enable_2fa']) {
417            $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to disable 2FA on a forced account: {$user->user_email}", (int) $id, 'User');
418            return response()->json(['message' => 'Two-factor authentication cannot be disabled on this account - it is enforced.'], 422);
419        }
420
421        $validated['enable_2fa'] = $isForced ? true : ($validated['enable_2fa'] ?? $user->enable_2fa);

423        // ── two_fa_forced_by_admin (sysadmin override) - see refactor-note in the header ──
424        if (array_key_exists('two_fa_forced_by_admin', $validated)) {
425            if (!$this->actorIsSysadmin($request)) {
426                $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to change 2FA enforcement (not sysadmin): {$user->user_email}", (int) $id, 'User');
427                return response()->json(['message' => 'Only sysadmin may change 2FA enforcement.'], 403);
428            }
429
430            if ($forcedByRole) {
431                return response()->json(['message' => '2FA is automatically enforced for this role - a manual override is unnecessary and not allowed.'], 422);
432            }
433        } else {
434            // Key was not present in the request at all - leave the existing value unchanged.
435            unset($validated['two_fa_forced_by_admin']);
436        }
437
438        // ── is_blocked (account lock) - ABSOLUTE protection for sysadmin ───────────────
439        // Checked BEFORE update() - we need to decide this before anything is written.
440        if (array_key_exists('is_blocked', $validated) && $validated['is_blocked']) {
441            if ((int) $request->user()->id === (int) $user->id) {
442                return response()->json(['message' => 'You cannot block your own account.'], 422);
443            }
444
445            // Effective role after any change in this request (same logic as
446            // $effectiveRoleId above, but explicitly through the role name for
447            // readability).
448            $effectiveRoleName = $effectiveRoleId ? CoreRole::find($effectiveRoleId)?->role_name : null;
449            $currentRoleNames = $user->roles()->pluck('role_name');
450
451            $isProtectedRole = in_array($effectiveRoleName, self::NEVER_BLOCK_ROLE_NAMES, true)
452                || $currentRoleNames->intersect(self::NEVER_BLOCK_ROLE_NAMES)->isNotEmpty();
453
454            if ($isProtectedRole) {
455                $this->logAction($request, CoreLog::class, 'update_denied', 'User', "Denied attempt to block a protected account: {$user->user_email}", (int) $id, 'User');
456                return response()->json(['message' => 'Accounts with the sysadmin role can never be blocked.'], 422);
457            }
458        }
459
460        $wasBlocked = (bool) $user->is_blocked;
461
462        // permission_ids is handled separately via applyExplicitPermissions() - strip
463        // it out of the mass-assignable payload before $user->update().
464        $requestedPermissionIds = array_key_exists('permission_ids', $validated) ? $validated['permission_ids'] : null;
465        unset($validated['permission_ids']);
466
467        // A password is NEVER written by update() - see header refactor-note
468        // (2026-10-07). The rule was removed from UpdateUserRequest; this unset is a
469        // second line of defence in case the key ever reappears in the validated data.
470        unset($validated['user_password_hash']);
471
472        DB::beginTransaction();
473        try {
474            $user->update($validated);
475
476            if (isset($validated['role_id'])) {
477                if ($request->user()->id !== $user->id) {
478                    $user->roles()->sync([$validated['role_id']]);
479                }
480            }
481
482            // ── Role-change wipe - see header refactor-note (2026-09-06). MUST
483            // happen before applyExplicitPermissions() reads the account's current
484            // explicit grants, otherwise it would treat pre-change grants as
485            // "outside the actor's authority, leave untouched" and effectively
486            // undo the wipe.
487            if ($roleChanged) {
488                $user->explicitPermissions()->sync([]);
489                $this->logAction(
490                    $request,
491                    CoreLog::class,
492                    'update',
493                    'User',
494                    "Role changed for {$user->user_email} - all explicit permission grants were reset to zero.",
495                    $user->id,
496                    'User'
497                );
498            }
499
500            // @bugfix-note (2026-09-06) Uses $effectiveRoleId directly (already
501            // resolved above to reflect the role the account will ACTUALLY end up
502            // with, self-edit-aware) instead of re-preferring $validated['role_id'] -
503            // the previous `$validated['role_id'] ?? $effectiveRoleId` expression
504            // would silently reintroduce the same self-edit bug this whole patch
505            // fixes, by feeding a never-applied role_id into the redundant-
506            // permission filtering inside applyExplicitPermissions().
507            if ($requestedPermissionIds !== null) {
508                $this->applyExplicitPermissions($request, $user, $requestedPermissionIds, (int) $effectiveRoleId);
509            }
510
511            // Transition false -> true: IMMEDIATELY invalidate all active access,
512            // otherwise a compromised account could keep using the system until its
513            // access token expires on its own (see backlog decision on the purpose of
514            // blocking).
515            if (!$wasBlocked && $user->is_blocked) {
516                $user->tokens()->delete();
517                RefreshToken::where('user_id', $user->id)->delete();
518                $this->logAction($request, CoreLog::class, 'account_blocked', 'User', "Account blocked, active tokens invalidated: {$user->user_email}", $user->id, 'User');
519            }
520
521            DB::commit();
522            $this->logAction($request, CoreLog::class, 'update', 'User', "User updated: {$user->user_email}", $user->id, 'User');
523            return response()->json(new UserResource($user->load(['roles.permissions', 'explicitPermissions'])));
524
525        } catch (\Exception $e) {
526            DB::rollBack();
527            $this->logAction($request, CoreLog::class, 'error', 'User', "Error updating user ID {$id}: " . $e->getMessage(), (int) $id, 'User');
528            return response()->json(['message' => 'Server error while saving.'], 500);
529        }

582

584                $request,
585                CoreLog::class,
586                'PasswordChanged',
587                'User',
588                "Password changed for: {$user->user_email} " . (!$isOwner ? "(performed by sysadmin: {$auth->user_email})" : ""),
589                $user->id,
590                'User'
591            );
592
593            $this->notifyPasswordChanged($request, $user);
594
595            return response()->json(['message' => 'Password changed successfully.']);
596        } catch (\Exception $e) {
597            $this->logAction($request, CoreLog::class, 'error', 'User', "Error changing password for ID {$id}: " . $e->getMessage(), (int) $id, 'User');
598            return response()->json(['message' => 'Password change failed.'], 500);
599        }

749    /**
750     * @description Applies an explicit-permission request (`permission_ids`) to a
751     * user's `user_permissions` pivot, protected by an anti-privilege-escalation
752     * guard.
753     *
754     * SECURITY: an actor may only grant or revoke explicit permissions that they
755     * THEMSELVES effectively hold (role ∪ their own explicit grants) - see
756     * `User::getPermissionsAttribute()`. Sysadmin is exempt, matching how
757     * `CheckPermission` middleware already treats sysadmin everywhere else. Without
758     * this guard, any account with `core-administrators-update` could hand out
759     * permissions it does not itself possess - a direct privilege-escalation path -
760     * so this check is not optional and cannot be skipped by the caller.
761     *
762     * Permissions the TARGET's role already grants are silently filtered out before
763     * saving: explicit grants only make sense for permissions the role does NOT
764     * already carry (project decision log - "K roli půjde uživateli přidat volitelné
765     * explicitní permissions navíc - jen ty, které role sama nemá").

770     * slice of permission-space they themselves control. (Note: on an actual ROLE
771     * CHANGE, `update()` wipes the pivot to empty BEFORE calling this method - see
772     * header refactor-note (2026-09-06) - so this "leave untouched" behaviour only
773     * ever applies when the role is NOT changing.)

776     * @param User $target The user whose explicit permissions are being updated.

778     * @param int|null $targetRoleId The role_id the target will end up with, used to filter out redundant grants.
779     * @return void
780     */
781    private function applyExplicitPermissions(Request $request, User $target, array $requestedPermissionIds, ?int $targetRoleId): void
782    {
783        $actor = $request->user();
784        $actorIsSysadmin = $this->actorIsSysadmin($request);
785
786        // The set of permission ids the actor is allowed to touch. Sysadmin may touch
787        // anything; everyone else is limited to permission keys they themselves
788        // effectively hold.
789        if ($actorIsSysadmin) {
790            $allowedPermissionIds = CorePermission::pluck('id')->all();
791        } else {
792            $actorPermissionKeys = $actor->permissions ?? [];
793            $allowedPermissionIds = CorePermission::whereIn('permission_key', $actorPermissionKeys)->pluck('id')->all();
794        }
795
796        // Permissions the target's role already carries - explicit grants for these
797        // would be redundant, so they are filtered out rather than stored.
798        $rolePermissionIds = $targetRoleId
799            ? CoreRole::find($targetRoleId)?->permissions()->pluck('core_permissions.id')->all() ?? []
800            : [];
801
802        $requestedWithinAuthority = array_values(array_intersect($requestedPermissionIds, $allowedPermissionIds));
803        $requestedFinal = array_values(array_diff($requestedWithinAuthority, $rolePermissionIds));
804
805        $existingExplicitIds = $target->explicitPermissions()->pluck('core_permissions.id')->all();
806        $outsideActorAuthority = array_values(array_diff($existingExplicitIds, $allowedPermissionIds));
807
808        // Final pivot state = (existing grants outside the actor's authority, left
809        // untouched) UNION (whatever the actor is requesting, within their authority
810        // and not already covered by the role).
811        $finalIds = array_values(array_unique(array_merge($outsideActorAuthority, $requestedFinal)));
812
813        $syncData = [];
814        foreach ($finalIds as $permissionId) {
815            $syncData[$permissionId] = ['granted_by' => $actor->id];
816        }
817
818        $target->explicitPermissions()->sync($syncData);
819
820        $this->logAction(
821            $request,
822            CoreLog::class,
823            'update',
824            'User',
825            "Explicit permissions updated for: {$target->user_email} (" . count($finalIds) . ' active grant(s))',
826            $target->id,
827            'User'
828        );
829    }
