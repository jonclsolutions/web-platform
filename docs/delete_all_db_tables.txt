-- @description Drops every table of the database you are connected to in
--   phpMyAdmin / the mysql client - DATABASE() returns the name of the
--   currently selected database, no name is hard-coded.
--
--   Without PREPARE/EXECUTE (unreliable in phpMyAdmin) - a plain SELECT
--   generates the DROP TABLE statements, which you copy and run as the
--   second step.
--
--   WARNING: irreversible. Make a backup (Export) BEFORE running it.
-- @dependencies none
-- ═══════════════════════════════════════════════════════════════


-- ── STEP 1: generate the DROP statements (nothing is deleted) ─────
SELECT CONCAT('DROP TABLE IF EXISTS `', TABLE_NAME, '`;') AS drop_statement
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = DATABASE()
ORDER BY TABLE_NAME;
