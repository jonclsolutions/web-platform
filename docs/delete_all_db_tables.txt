-- ── KROK 1: vygenerování DROP příkazů (nic se nesmaže) ───────────
-- ── !!!!!!!!!!!!! ZNENIT NAZEV DB !!!!!!!!!!!!!§
SELECT CONCAT('DROP TABLE IF EXISTS `', TABLE_NAME, '`;') AS drop_statement
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = 'f191121'
ORDER BY TABLE_NAME;


-- ── KROK 2: skutečné smazání ──────────────────────────────────────
-- Zkopírujte VÝSTUP z KROKU 1 (celý sloupec drop_statement) a vložte
-- ho místo komentáře níže, pak spusťte celý tenhle blok najednou.

SET FOREIGN_KEY_CHECKS = 0;
